\noindent
\textbf{\large LISTA DE SÍMBOLOS}

\begin{listsymbol}
	\item[$H$] Función de hash criptográfica.
	\item[$\Vert$] Concatenación de cadenas de bits.
	\item[$\sigma$] Estado global de Ethereum.
	\item[$\Upsilon$] Función de transición de estado de Ethereum.
	\item[$T$] Transacción de Ethereum.
	\item[$g_{\mathrm{lim}}$] Límite de gas declarado por una transacción.
	\item[$g_{\mathrm{usado}}$] Gas efectivamente consumido por una transacción.
	\item[$g_{\mathrm{bloque}}$] Gas total consumido por un bloque.
	\item[$g_{\mathrm{obj}}$] Gas objetivo por bloque según EIP-1559.
	\item[$p_{\mathrm{base}}$] Tarifa base por unidad de gas.
	\item[$p'_{\mathrm{base}}$] Tarifa base por unidad de gas del bloque siguiente.
	\item[$p_{\mathrm{prop}}$] Propina efectiva por unidad de gas.
	\item[$p_{\max}$] Precio máximo por unidad de gas declarado por el emisor.
	\item[$p^{\mathrm{prop}}_{\max}$] Propina máxima por unidad de gas declarada por el emisor.
	\item[$\mathcal{C}$] Comité de sincronización de la \textit{Beacon Chain}.
	\item[$\mathcal{C}'$] Subconjunto de miembros del comité de sincronización que participaron en una firma.
	\item[$\mathit{pk}_i$] Clave pública BLS del miembro $i$ del comité de sincronización.
	\item[$\mathit{pk}_{\mathrm{agg}}$] Clave pública BLS agregada.
	\item[$\mathit{sig}$] Firma BLS agregada.
	\item[$\mathbb{G}_1$] Primer grupo de origen del emparejamiento bilineal.
	\item[$\mathbb{G}_2$] Segundo grupo de origen del emparejamiento bilineal.
	\item[$\mathbb{G}_T$] Grupo de destino del emparejamiento bilineal.
	\item[$e$] Emparejamiento bilineal $e\colon \mathbb{G}_1 \times \mathbb{G}_2 \to \mathbb{G}_T$.
	\item[$g_1$] Generador del grupo $\mathbb{G}_1$.
	\item[$H_{\mathbb{G}_2}$] Función de hash que transforma mensajes en puntos de $\mathbb{G}_2$.
	\item[$v$] Valor almacenado en una hoja de un árbol de Merkle indexado.
	\item[$i_{\mathrm{sig}}$] Índice de la hoja que contiene el siguiente valor mayor en un árbol de Merkle indexado.
	\item[$v_{\mathrm{sig}}$] Siguiente valor mayor almacenado en un árbol de Merkle indexado.
	\item[$\mathcal{P}$] Probador (\textit{prover}) de un sistema de prueba.
	\item[$\mathcal{P}^{*}$] Probador potencialmente deshonesto.
	\item[$\mathcal{V}$] Verificador (\textit{verifier}) de un sistema de prueba.
	\item[$x$] Enunciado (\textit{statement}); entrada pública.
	\item[$w$] Testigo (\textit{witness}); entrada privada del probador.
	\item[$\pi$] Prueba (\textit{proof}) generada por el probador.
	\item[$\mathcal{R}$] Relación que define los pares válidos de enunciado y testigo.
	\item[$\mathbb{F}_p$] Cuerpo finito de $p$ elementos, con $p$ primo.
	\item[$A, B, C$] Combinaciones lineales de señales en una restricción cuadrática $A \cdot B + C = 0$ de Circom.
\end{listsymbol}
